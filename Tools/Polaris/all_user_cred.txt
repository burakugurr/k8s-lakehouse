echo "       POLARIS YENİ CREDENTIAL ÜRETİMİ (RECREATE)"
echo "=========================================================="

for user in burak ahmet aleyna zehra; do
  echo ">>> Kullanıcı: $user"
  
  # 1. Varsa eski principal'ı sil
  kubectl exec -it $TRINO_POD -n bigdata -- curl -s -X DELETE \
    http://polaris-catalog-service.bigdata.svc.cluster.local:8181/api/management/v1/principals/$user \
    -H "Authorization: Bearer $ADMIN_TOKEN" > /dev/null

  # 2. Yeniden oluştur ve dönen JSON'ı göster
  RESP=$(kubectl exec -it $TRINO_POD -n bigdata -- curl -s -X POST \
    http://polaris-catalog-service.bigdata.svc.cluster.local:8181/api/management/v1/principals \
    -H "Authorization: Bearer $ADMIN_TOKEN" \
    -H "Content-Type: application/json" \
    -d "{\"principal\":{\"name\":\"$user\"}}")

  echo "Oluşturma Yanıtı:"
  echo "$RESP"
  echo "----------------------------------------------------------"
done

echo "--> Roller kullanıcılara atanıyor..."
